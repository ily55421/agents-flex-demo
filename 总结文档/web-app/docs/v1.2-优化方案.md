## 3. PWA 支持（优先级：高）

### 3.1 Service Worker 离线缓存

**问题**：无网络时无法访问，每次刷新都重新请求资源。

**方案**：
- 注册 Service Worker，缓存核心资源（HTML/CSS/JS/字体）
- 实现 Cache-First 策略：优先从缓存读取，后台更新
- 离线页面：无网络时显示友好的离线提示
- 缓存版本管理：新版本发布时自动清理旧缓存

```javascript
// sw.js
const CACHE_NAME = 'docblog-v1.2';
const STATIC_ASSETS = [
  '/',
  '/index.html',
  '/css/style.css',
  '/css/prism-tomorrow.css',
  '/js/app.js',
  '/js/storage.js',
  '/js/theme.js',
  '/js/router.js',
  '/js/utils.js',
  '/js/ui.js',
  '/js/prism.min.js'
];

self.addEventListener('install', (e) => {
  e.waitUntil(
    caches.open(CACHE_NAME).then(cache => cache.addAll(STATIC_ASSETS))
  );
});

self.addEventListener('fetch', (e) => {
  e.respondWith(
    caches.match(e.request).then(response => {
      return response || fetch(e.request);
    })
  );
});
```

## 4. 搜索增强（优先级：高）

### 4.1 高级搜索语法

**问题**：仅支持简单关键词搜索，无法满足复杂查询需求。

**方案**：
- 支持布尔运算符：`Java AND Spring`（同时包含）
- 支持排除语法：`Java NOT Spring`（排除 Spring）
- 支持短语精确匹配：`"Spring Boot"`（完整短语）
- 支持分类限定：`category:Java 微服务`
- 支持标签限定：`tag:springboot`
- 前端搜索框添加语法提示下拉

```javascript
// 搜索语法解析器
function parseSearchQuery(query) {
  const result = {
    include: [],
    exclude: [],
    phrase: [],
    category: null,
    tag: null,
    raw: query
  };
  
  // 匹配精确短语 "..."
  const phraseRegex = /"([^"]+)"/g;
  let match;
  while ((match = phraseRegex.exec(query)) !== null) {
    result.phrase.push(match[1]);
  }
  query = query.replace(phraseRegex, '');
  
  // 匹配分类限定 category:xxx
  const categoryMatch = query.match(/category:(\S+)/i);
  if (categoryMatch) {
    result.category = categoryMatch[1];
    query = query.replace(categoryMatch[0], '');
  }
  
  // 匹配标签限定 tag:xxx
  const tagMatch = query.match(/tag:(\S+)/i);
  if (tagMatch) {
    result.tag = tagMatch[1];
    query = query.replace(tagMatch[0], '');
  }
  
  // 解析剩余关键词（AND/OR/NOT）
  const tokens = query.trim().split(/\s+/);
  for (let i = 0; i < tokens.length; i++) {
    if (tokens[i].toUpperCase() === 'NOT' && i + 1 < tokens.length) {
      result.exclude.push(tokens[i + 1]);
      i++;
    } else if (tokens[i]) {
      result.include.push(tokens[i]);
    }
  }
  
  return result;
}
```

```javascript
// 阅读统计
const ReadingStats = {
  record(docId, duration) {
    const today = new Date().toISOString().slice(0, 10);
    const key = `reading_${today}`;
    const stats = Storage.get(key, { docs: [], totalTime: 0 });
    
    if (!stats.docs.includes(docId)) {
      stats.docs.push(docId);
    }
    stats.totalTime += duration;
    Storage.set(key, stats);
  },
  
  getWeeklyStats() {
    const stats = [];
    for (let i = 6; i >= 0; i--) {
      const date = new Date();
      date.setDate(date.getDate() - i);
      const key = `reading_${date.toISOString().slice(0, 10)}`;
      const dayStats = Storage.get(key, { docs: [], totalTime: 0 });
      stats.push({
        date: date.toISOString().slice(0, 10),
        docs: dayStats.docs.length,
        minutes: Math.round(dayStats.totalTime / 60)
      });
    }
    return stats;
  }
};
```

**问题**：前端报错无法及时发现和定位。

**方案**：
- 全局 `window.onerror` 和 `window.onunhandledrejection` 捕获
- 错误信息收集：错误类型、堆栈、用户代理、当前路由
- 错误上报到后端 `/api/log` 接口
- 后端记录到 `frontend_errors` 表
- 管理后台错误看板：错误趋势、高频错误、影响用户

```javascript
// 前端错误监控
window.addEventListener('error', (e) => {
  reportError({
    type: 'javascript',
    message: e.message,
    filename: e.filename,
    lineno: e.lineno,
    colno: e.colno,
    stack: e.error?.stack,
    url: window.location.href,
    userAgent: navigator.userAgent,
    timestamp: Date.now()
  });
});

window.addEventListener('unhandledrejection', (e) => {
  reportError({
    type: 'promise',
    message: e.reason?.message || String(e.reason),
    stack: e.reason?.stack,
    url: window.location.href,
    timestamp: Date.now()
  });
});

function reportError(errorInfo) {
  fetch('/api/log', {
    method: 'POST',
    headers: { 'Content-Type': 'application/json' },
    body: JSON.stringify(errorInfo)
  }).catch(() => {});
}
```

### 6.2 性能指标上报

**方案**：
- 使用 Performance API 采集核心指标
- FCP（First Contentful Paint）
- LCP（Largest Contentful Paint）
- FID（First Input Delay）
- CLS（Cumulative Layout Shift）
- 上报到后端，管理后台展示性能趋势

```javascript
// 性能指标采集
function collectPerformanceMetrics() {
  const metrics = {};
  
  // FCP
  new PerformanceObserver((list) => {
    const entry = list.getEntries()[0];
    metrics.fcp = entry.startTime;
  }).observe({ entryTypes: ['paint'] });
  
  // LCP
  new PerformanceObserver((list) => {
    const entries = list.getEntries();
    const lastEntry = entries[entries.length - 1];
    metrics.lcp = lastEntry.startTime;
  }).observe({ entryTypes: ['largest-contentful-paint'] });
  
  // CLS
  let cls = 0;
  new PerformanceObserver((list) => {
    for (const entry of list.getEntries()) {
      if (!entry.hadRecentInput) {
        cls += entry.value;
      }
    }
    metrics.cls = cls;
  }).observe({ entryTypes: ['layout-shift'] });
  
  // 页面卸载时上报
  window.addEventListener('beforeunload', () => {
    navigator.sendBeacon('/api/performance', JSON.stringify(metrics));
  });
}
```

### 9.2 高级搜索后端实现

```python
# server.py
class DocRequestHandler(http.server.SimpleHTTPRequestHandler):
    def do_GET(self):
        # ...
        if path == '/api/search':
            return self._api_search(qs)
        # ...
    
    def _api_search(self, qs):
        query = qs.get('q', [''])[0].strip()
        category = qs.get('category', [''])[0]
        page = int(qs.get('page', ['1'])[0])
        limit = int(qs.get('limit', [str(CONFIG['page_limit'])])[0])
        
        if not query:
            return self._json_response({"results": [], "total": 0, "page": 1, "pages": 0})
        
        # 解析高级搜索语法
        parsed = self._parse_search_query(query)
        
        # 构建 FTS5 查询
        fts_query = self._build_fts_query(parsed)
        
        # 执行搜索
        conn = get_db()
        cursor = conn.cursor()
        
        # 基础查询
        base_sql = """
            SELECT d.id, d.title, d.category_name, d.word_count, d.file_mtime,
                   snippet(fts_docs, 0, '<mark>', '</mark>', '...', 32) as snippet,
                   rank
            FROM fts_docs
            JOIN documents d ON fts_docs.rowid = d.id
            WHERE fts_docs MATCH ?
        """
        params = [fts_query]
        
        # 分类过滤
        if category:
            base_sql += " AND d.category_slug = ?"
            params.append(category)
        
        # 标签过滤
        if parsed.get('tag'):
            base_sql += """ AND d.id IN (
                SELECT document_id FROM document_tags 
                JOIN tags ON document_tags.tag_id = tags.id 
                WHERE tags.slug = ?
            )"""
            params.append(parsed['tag'])
        
        # 排序和分页
        base_sql += " ORDER BY rank LIMIT ? OFFSET ?"
        params.extend([limit, (page - 1) * limit])
        
        cursor.execute(base_sql, params)
        results = [dict(row) for row in cursor.fetchall()]
        
        # 统计总数
        count_sql = "SELECT COUNT(*) FROM fts_docs WHERE fts_docs MATCH ?"
        count_params = [fts_query]
        if category:
            count_sql += " AND d.category_slug = ?"
            count_params.append(category)
        
        cursor.execute(count_sql, count_params)
        total = cursor.fetchone()[0]
        
        return self._json_response({
            "results": results,
            "total": total,
            "page": page,
            "pages": (total + limit - 1) // limit,
            "query": query,
            "parsed": parsed
        })
    
    def _parse_search_query(self, query):
        """解析高级搜索语法"""
        result = {"include": [], "exclude": [], "phrase": [], "category": None, "tag": None}
        
        # 精确短语 "..."
        import re
        phrases = re.findall(r'"([^"]+)"', query)
        result["phrase"] = phrases
        query = re.sub(r'"[^"]*"', '', query)
        
        # 分类限定 category:xxx
        cat_match = re.search(r'category:(\S+)', query, re.I)
        if cat_match:
            result["category"] = cat_match.group(1)
            query = query.replace(cat_match.group(0), '')
        
        # 标签限定 tag:xxx
        tag_match = re.search(r'tag:(\S+)', query, re.I)
        if tag_match:
            result["tag"] = tag_match.group(1)
            query = query.replace(tag_match.group(0), '')
        
        # 解析布尔运算符
        tokens = query.strip().split()
        i = 0
        while i < len(tokens):
            if tokens[i].upper() == 'NOT' and i + 1 < len(tokens):
                result["exclude"].append(tokens[i + 1])
                i += 2
            elif tokens[i]:
                result["include"].append(tokens[i])
                i += 1
            else:
                i += 1
        
        return result
    
    def _build_fts_query(self, parsed):
        """构建 FTS5 查询字符串"""
        parts = []
        
        # 包含关键词
        for term in parsed["include"]:
            parts.append(term)
        
        # 精确短语
        for phrase in parsed["phrase"]:
            parts.append(f'"{phrase}"')
        
        # 排除关键词
        for term in parsed["exclude"]:
            parts.append(f'NOT {term}')
        
        return ' AND '.join(parts) if parts else '*'
```
